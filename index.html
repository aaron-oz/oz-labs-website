# Spatial Gaussian Process
# Squared Exponential Kernel

kernel <- function(x1, x2, l, sigma) {
  d <- sqrt(sum((x1 - x2)^2))
  sigma^2 * exp(-d^2 / (2 * l^2))
}

# Build covariance matrix
K <- outer(locations, locations,
           Vectorize(kernel),
           l = 0.5, sigma = 1.0)

# Sample from GP prior
f <- mvrnorm(n = 1,
             mu = rep(0, n),
             Sigma = K)
